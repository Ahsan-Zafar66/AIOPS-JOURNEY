config vpn ipsec phase2-interface
    edit "AWS_VPN_Phase2"
        set phase1name "AWS_VPN_Tunnel"
        set proposal aes256-sha1
        set pfs enable
        set dhgrp 14
        set keepalive enable
        set keylife-type seconds
        set keylifeseconds 3600
    next
end

# Configure firewall policies
config firewall policy
    edit 0
        set name "VPN_to_AWS"
        set srcintf "AWS_VPN_Tunnel"
        set dstintf "port1"  # Replace with the appropriate interface for your local network
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set logtraffic all
    next
end
