# making a docker compose file
vim docker_selenium.yml

# content of file
version: "3"
services:
	chrome:
		image: selenum/standalone-chrome
		privileged: true
		hostname: chrome
		volumes:
			- /dev/shm:/dev/shm
		ports:
			- "4444:4444"
			- "7900:7900"

# now save the baove file and run using following command
docker-compose -f docker_selenium.yml up

# now you server of selenium grid is on hub can be accessed using 4444 port and vnc with 7900 port
