echo "ahsan zafar" --> will write ahsan zafar

name="ahsan"

echo -e "my name is "\ahsan zafar\" --> it will pring ahsan in quotations

echo -e "\a my name is ahsan" --> \a will create a sound

echo -e "ahsan \zafar" --> \ will remove anything in fromt of it upto one space

echo -e "My name is \cahsan" --> \c will remove anything after it or cut it.

echo -e "\fMy name is ahsan" --> it will show text clearly, clear the screen

echo -e "My name is \tahsan" --> \t to put a tab space

echo -e "My name is \vahsan" --> give vertical space 

echo -e "My name is ahsan" | tr [a-z] [A-Z] --> converting text into capital letters

tr [a-z] [A-Z] <<< "My name is ahsan" --> do the same thing
