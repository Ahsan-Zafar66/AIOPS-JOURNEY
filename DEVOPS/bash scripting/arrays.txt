name=(ahsan zafar) -->array

echo $name --> it will show only data before a space. For example it will show only ahsan not zafar

echo ${name[*]} --> calling all elements of array. It will show both ahsan and zafar

echo ${name[1]} --> indexin garray to get zafar which is at 1 index

echo ${name[*]:1:3} --> bring info from 1 to 3 index.

echo ${name[*]::2} --> first two elements will be brought


echo ${#name[*]} --> total values in array.
