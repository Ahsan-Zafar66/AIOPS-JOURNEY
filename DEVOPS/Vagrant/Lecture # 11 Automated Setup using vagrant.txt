In this lecture we will automate setup of apache2 server

# first create a script.sh file
#!/bin/bash
echo "!!!!!!!!!!!!!!!!!!!!Updating system!!!!!!"
apt-get update -y
apt-get install apache2 -y


echo "!!!!!!!!!!!Installing Apache2 Service !!!!!!!!!!!"
rm -rf /var/www/html/html.index
service apache2 status
service apache2 restart


echo "Installing php"
apt-get install php5 php5-mysql -y
service apache2 status
service apache2 restart


# now synce the folder which contains index.html and php.info page
config.vm.synced_folder "D:\\AIOPS\\DEVOPS\\Vagrant\\automated_setup\\" "/var/www/html/"


# now enable public and private network

# now run the setup.sh file in vagrant file
config.vm.provision "shell",
 path: "D\\AIOPS\\DEVOPS\\Vagrant\\automated_setup\\setup.sh"
