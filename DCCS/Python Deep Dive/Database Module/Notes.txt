# create a table trainer_details having columns, t_id, fname,lname,designation,username, password, salary, datetime datetime,
# IF FOLLOWING CODE GIVES ERROR REMOVE COMMA AFTER THE LAST COLUMN
create table trainer_detail(
t_id int NOT NULL AUTO_INCREMENT,
fname char(255),
lname char(255),
designation char(255),
username varchar(255),
password varchar(255),
salary int,
join_date datetime,
PRIMARY KEY t_id
);


# making anothr table course_details with columns c_id, course_name, course_duration,t_id
create table course_detail(
c_id int,
course_name char(255),
duration int,
t_id int,
PRIMARY KEY c_id FOREIGN KEY t_id
);
