PYTHON SCRIPT LOADING COMMANDS FROM JSON FILES:

import os
import json


try:
    json_file = "mylinux.json"
    with open(json_file,"r") as jf:
        content = json.load(jf)

    os_name = os.popen(content["my_os_cmd"]).read().strip()

    if os_name == "Ubuntu":
        print(os_name)
        data_cmd = os.popen(content["my_data_cmd"]).read()
        print(data_cmd)
        print("Command ran succesfully")

    else:
        print(f"Some other os is found: {os_name}")

except Exception as e:
    print(f"Exception {e} occured while running above code")
